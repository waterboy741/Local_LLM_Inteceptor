cmake_minimum_required(VERSION 3.24)
project(llm_interceptor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(INTERCEPTOR_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(INTERCEPTOR_SANITIZERS)
    include(CheckCXXSourceCompiles)
    set(CMAKE_REQUIRED_FLAGS "-fsanitize=address,undefined")
    check_cxx_source_compiles("int main() { return 0; }" SANITIZERS_SUPPORTED)
    unset(CMAKE_REQUIRED_FLAGS)
    if(NOT SANITIZERS_SUPPORTED)
        message(FATAL_ERROR "Compiler/linker do not support requested sanitizers")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Widgets Network HttpServer)
add_library(interceptor_core src/core/SessionConfig.cpp src/session/SessionController.cpp src/session/SessionController.hpp)
target_include_directories(interceptor_core PUBLIC src)
target_link_libraries(interceptor_core PUBLIC Qt6::Core Qt6::Network)
add_library(interceptor_network src/network/ArpCodec.cpp src/network/NetworkSession.cpp src/network/NetworkSession.hpp src/network/HelperDispatcher.cpp src/network/UnixPeerCredentials.cpp src/network/UnixPeerCredentials.hpp src/network/UnixSocketCommandTransport.cpp src/network/UnixSocketCommandTransport.hpp src/network/LabPreflight.cpp src/network/LabPreflight.hpp src/network/PfDiagnostics.cpp src/network/PfDiagnostics.hpp src/network/RouteNeighborDiagnostics.cpp src/network/RouteNeighborDiagnostics.hpp src/network/PfAnchorOperations.cpp src/network/PfAnchorOperations.hpp src/network/PfAnchorQProcessRunner.cpp src/network/PfAnchorQProcessRunner.hpp)
target_link_libraries(interceptor_network PUBLIC interceptor_core Qt6::Network)
add_executable(lab_preflight src/lab_preflight_main.cpp)
target_link_libraries(lab_preflight PRIVATE interceptor_network Qt6::Core)
add_library(interceptor_proxy src/proxy/ProxyServer.cpp src/proxy/ProxyServer.hpp)
target_link_libraries(interceptor_proxy PUBLIC interceptor_core Qt6::Network Qt6::HttpServer)
add_library(interceptor_inspection src/inspection/InspectionArchive.cpp src/inspection/InspectionArchive.hpp)
target_link_libraries(interceptor_inspection PUBLIC interceptor_proxy Qt6::Core)
add_library(interceptor_gui src/gui/MainWindow.cpp src/gui/MainWindow.hpp)
target_link_libraries(interceptor_gui PUBLIC interceptor_core interceptor_proxy interceptor_inspection Qt6::Widgets Qt6::Network)
foreach(target interceptor_core interceptor_network interceptor_proxy interceptor_inspection interceptor_gui)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(llm_interceptor MACOSX_BUNDLE src/main.cpp)
target_link_libraries(llm_interceptor PRIVATE interceptor_gui)
set_target_properties(llm_interceptor PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "llm_interceptor")
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()
